class ElementDouble {
  constructor(tag, attrs, document) {
    this.tag = tag;
    this.attrs = attrs;
    this.document = document;
    this.children = [];
    this.listeners = new Map();
    this._text = '';
    this._value = undefined;
    this.hidden = Object.hasOwn(attrs, 'hidden');
    this.disabled = Object.hasOwn(attrs, 'disabled');
    this.dataset = Object.fromEntries(Object.entries(attrs).filter(([key]) => key.startsWith('data-')).map(([key, value]) => [key.slice(5).replace(/-([a-z])/g, (_, letter) => letter.toUpperCase()), value]));
    const classes = new Set((attrs.class || '').split(/\s+/));
    this.classList = {
      contains: name => classes.has(name),
      toggle: (name, enabled) => enabled ? classes.add(name) : classes.delete(name)
    };
    const properties = new Map();
    this.style = { setProperty: (key, value) => properties.set(key, value), getPropertyValue: key => properties.get(key) };
  }
  get textContent() { return this._text + this.children.map(child => child.textContent).join(''); }
  set textContent(value) { this._text = String(value); this.children = []; }
  get value() {
    if (this._value !== undefined) return this._value;
    if (this.tag === 'select') return this.querySelector('option')?.value || '';
    return this.attrs.value ?? this.textContent;
  }
  set value(value) { this._value = String(value); }
  setAttribute(name, value) { this.attrs[name] = String(value); }
  getAttribute(name) { return this.attrs[name] ?? null; }
  addEventListener(name, handler) {
    if (!this.listeners.has(name)) this.listeners.set(name, []);
    this.listeners.get(name).push(handler);
  }
  emit(name, fields = {}) {
    for (const handler of this.listeners.get(name) || []) handler({ type: name, target: this, ...fields });
  }
  click() { if (!this.disabled) this.emit('click'); }
  change(value) { this.value = value; this.emit('change'); }
  focus() { this.document.activeElement = this; }
  matches(selector) {
    if (selector.startsWith('.')) return this.classList.contains(selector.slice(1));
    const attr = selector.match(/^\[([\w-]+)(?:="([^"]*)")?\]$/);
    if (attr) return Object.hasOwn(this.attrs, attr[1]) && (attr[2] === undefined || this.attrs[attr[1]] === attr[2]);
    return this.tag === selector;
  }
  querySelectorAll(selector) {
    return this.children.flatMap(child => [...(child.matches(selector) ? [child] : []), ...child.querySelectorAll(selector)]);
  }
  querySelector(selector) { return this.querySelectorAll(selector)[0] || null; }
}

function parseDocument() {
  const document = { activeElement: null };
  const tree = new ElementDouble('document', {}, document);
  const stack = [tree];
  const markup = html.replace(/<style>[\s\S]*?<\/style>|<script>[\s\S]*?<\/script>/g, '');
  for (const token of markup.match(/<[^>]+>|[^<]+/g) || []) {
    if (token.startsWith('</')) { stack.pop(); continue; }
    if (token.startsWith('<')) {
      const name = token.match(/^<([\w-]+)/)?.[1];
      if (!name) continue;
      const attrs = {};
      const rest = token.slice(name.length + 1, -1);
      for (const match of rest.matchAll(/([\w-]+)(?:="([^"]*)")?/g)) attrs[match[1]] = match[2] || '';
      const element = new ElementDouble(name, attrs, document);
      stack.at(-1).children.push(element);
      if (!['input', 'img', 'br', 'hr', 'meta', 'link'].includes(name)) stack.push(element);
    } else stack.at(-1)._text += token;
  }
  document.getElementById = id => tree.querySelectorAll('[id]').find(element => element.attrs.id === id) || null;
  return document;
}

const clone = value => JSON.parse(JSON.stringify(value));
function runtime(options = {}) {
  const document = parseDocument();
  const saves = [];
  const listeners = new Map();
  const tweaks = [];
  const window = {
    addEventListener: (name, handler) => {
      if (!listeners.has(name)) listeners.set(name, []);
      listeners.get(name).push(handler);
    },
    emit: (name, fields) => { for (const handler of listeners.get(name) || []) handler(fields); }
  };
  if (!options.absentHost) {
    window.openai = { widgetState: options.snapshot };
    if (!options.absentSave) window.openai.setWidgetState = payload => {
      saves.push(clone(payload));
      assert.ok(Buffer.byteLength(JSON.stringify(payload)) < 16384, 'snapshot exceeds 16 KiB');
      if (options.saveResult === 'throw') throw new Error('optional bridge unavailable');
      if (options.saveResult === 'reject') return Promise.reject(new Error('host rejected save'));
      if (options.saveResult === 'void') return undefined;
      return Promise.resolve();
    };
  }
  class TweakDouble {
    constructor(config) {
      this.config = config;
      this.controls = [];
      tweaks.push(this);
      config.onChange(); // An eager helper must not cause an initial snapshot write.
    }
    addSlider(bound, key, config) {
      this.controls.push({ bound, key, config });
      this.config.onChange();
    }
    change(key, value) {
      const control = this.controls.find(item => item.key === key);
      assert.ok(control, `missing design control ${key}`);
      control.bound[key] = value;
      this.config.onChange();
    }
  }
  const context = vm.createContext({ window, document, Promise, ...(options.noTweak ? {} : { Tweak: TweakDouble }) });
  vm.runInContext(script, context, { filename: 'TASK-072-compact-preview.html' });
  const root = document.getElementById('thaidlt-compact-preview');
  const apps = root.querySelectorAll('.td-app');
  assert.equal(apps.length, 2);
  assert.equal(root.querySelectorAll('[data-choice]').length, 8);
  return { document, window, saves, apps, tweaks, flow: name => apps.find(app => app.dataset.flow === name) };
}
const query = (app, selector) => {
  const found = app.querySelector(selector);
  assert.ok(found, `missing actual fixture element: ${selector}`);
  return found;
};
const action = (app, name) => query(app, `[data-action="${name}"]`);
const choice = (app, index) => query(app, `[data-choice="${index}"]`);
const late = (rt, widgetState) => rt.window.emit('openai:set_globals', { detail: { globals: { widgetState } } });
const flowDefaults = () => ({ index: 0, selected: null, answered: false, menu: false, vehicle: 'Car', topic: 'all', favorites: [false, false], design: { spacing: 16, radius: 12 } });
function sampleSnapshot() {
  return {
    privateContent: {
      preview: 'thaidlt-compact-preview', version: 1,
      flows: {
        direct: { index: 1, selected: 0, answered: true, menu: true, vehicle: 'Motorcycle', topic: 'lane', favorites: [false, true], design: { spacing: 20, radius: 4 } },
        confirm: { index: 0, selected: 2, answered: false, menu: false, vehicle: 'Car', topic: 'road', favorites: [true, false], design: { spacing: 12, radius: 16 } }
      }
    }
  };
}
function assertDefaults(app) {
  assert.equal(query(app, '[data-counter]').textContent, 'Sample question 1 of 2');
  assert.equal(query(app, '[data-topic]').value, 'all');
  assert.equal(query(app, '[data-vehicle]').value, 'Car');
  assert.equal(query(app, '.td-menu').hidden, true);
  assert.equal(query(app, '.td-feedback').hidden, true);
  assert.equal(action(app, 'favorite').getAttribute('aria-pressed'), 'false');
  assert.equal(app.style.getPropertyValue('--td-answer-space'), '16px');
  assert.equal(app.style.getPropertyValue('--td-radius'), '12px');
  if (app.dataset.flow === 'confirm') assert.equal(action(app, 'check').disabled, true);
}
